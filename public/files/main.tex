% Custom commands
%
% \resumeRow renders a left item and a right item as ONE horizontal box, so
% extraction keeps them on a single line and a role stays glued to its dates.
\newcommand{\resumeRow}[2]{\hbox to \linewidth{#1\hfill #2}}

\newcommand{\resumeItem}[1]{\item\small{#1}}

% \leavevmode is required: after \item TeX is in vertical mode, where an \hbox
% is appended as a whole line and the following \\ raises
% "LaTeX Error: There's no line here to end."
\newcommand{\resumeSubheading}[4]{%
	\vspace{4pt}\item\leavevmode
	\resumeRow{\textbf{#1}}{\small #2}\\
	\resumeRow{\textit{\small #3}}{\textit{\small #4}}
	\vspace{-2pt}}

% #1 project + role, #2 tech stack (own line, free to wrap), #3 date.
% Keeping the stack off the date line is what fixed the old collision where
% "...TurboRepo" ran straight into "June 2024" with no separator.
\newcommand{\resumeProjectHeading}[3]{%
	\vspace{4pt}\item\leavevmode
	\resumeRow{\small\textbf{#1}}{\small #3}\\
	{\small\textit{#2}}
	\vspace{-2pt}}
